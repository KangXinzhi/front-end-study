function quickSort(arr,left=0,right=arr.length-1){
    if(left>right){
        return
    }

    let index = partition(arr,left,right);
    quickSort(arr,index+1,right)
    quickSort(arr,left,index-1)

    return arr;
}

function partition(arr,left,right){
    let j = left;
    let num = arr[j];

    for(let i = left+1; i<=right; i++){
        if(arr[i]<=num){
            j++;
            swap(arr,i,j)
        }
    }
    swap(arr,left,j)
    return j;
}

function swap(arr,left,right){
    let temp = arr[left];
    arr[left] = arr[right];
    arr[right] = temp;
}

function quickSortInPlace(arr, left = 0, right = arr.length - 1) {
  if (left >= right) return;  // 如果左边索引不小于右边，表示只有一个元素，不需要排序

  const pivotIndex = partition(arr, left, right);  // 通过分区函数获取基准元素的位置
  quickSortInPlace(arr, left, pivotIndex - 1);  // 递归排序左半部分
  quickSortInPlace(arr, pivotIndex + 1, right);  // 递归排序右半部分
}

function partition(arr, left, right) {
  const pivot = arr[right];  // 选择最右边的元素作为基准
  let i = left;

  for (let j = left; j < right; j++) {
    if (arr[j] < pivot) {  // 如果当前元素小于基准，交换元素
      [arr[i], arr[j]] = [arr[j], arr[i]];  // 交换
      i++;
    }
  }

  [arr[i], arr[right]] = [arr[right], arr[i]];  // 最后将基准元素放到中间
  return i;  // 返回基准元素的位置
}
